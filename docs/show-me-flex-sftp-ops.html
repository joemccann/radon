Step 6 VPS after written grant

Pin host key. First list-dir. IPv4 only.

cat > /var/lib/radon/flex-secrets/ssh_config <<'EOF'
Host ibkr-flex
  HostName ftp2.interactivebrokers.com
  User <IBKR_SFTP_USER>
  Port 22
  IdentityFile /var/lib/radon/flex-secrets/ibkr_sftp
  IdentitiesOnly yes
  AddressFamily inet
  UserKnownHostsFile /var/lib/radon/flex-secrets/known_hosts
  StrictHostKeyChecking yes
